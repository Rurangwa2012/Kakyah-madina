rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function userPath() {
      return /databases/$(database)/documents/users/$(request.auth.uid);
    }

    function hasUserDoc() {
      return signedIn() && exists(userPath());
    }

    function userDoc() {
      return get(userPath()).data;
    }

    function isActiveUser() {
      return hasUserDoc()
        && (userDoc().active == true || userDoc().active == 'true');
    }

    function isOwner() {
      return isActiveUser() && (userDoc().role == 'owner' || userDoc().role == 'Owner');
    }

    function isCashier() {
      return isActiveUser() && (userDoc().role == 'cashier' || userDoc().role == 'Cashier');
    }

    function hasInventoryAccess() {
      return isOwner()
        || (isCashier()
          && (userDoc().inventory_access == true || userDoc().inventory_access == 'true'));
    }

    match /users/{userId} {
      allow get: if signedIn() && request.auth.uid == userId;
      allow list: if isOwner();
      allow create: if (signedIn()
          && request.auth.uid == userId
          && request.resource.data.role in ['owner', 'cashier']
          && request.resource.data.active == true)
        || (isOwner() && request.resource.data.role in ['owner', 'cashier']);
      allow update: if isOwner()
        || (isActiveUser()
          && request.auth.uid == userId
          && request.resource.data.role == resource.data.role
          && request.resource.data.active == resource.data.active
          && request.resource.data.inventory_access == resource.data.inventory_access
          && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['name', 'updated_at']));
      allow delete: if false;
    }

    match /menu/{itemId} {
      allow read: if isActiveUser();
      allow create, update, delete: if isOwner();
    }

    match /orders/{orderId} {
      allow read: if isOwner() || isCashier();
      allow create: if isActiveUser()
        && request.resource.data.cashier_id == request.auth.uid
        && request.resource.data.type == 'student'
        && request.resource.data.keys().hasAll([
          'order_number', 'lines', 'total_halalas', 'payment_method', 'status', 'date_key'
        ]);
      allow update: if isOwner();
      allow delete: if false;
    }

    match /group_orders/{orderId} {
      allow read: if isActiveUser();
      allow create: if isActiveUser()
        && request.resource.data.cashier_id == request.auth.uid;
      allow update: if isOwner()
        || (isCashier()
          && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['status', 'updated_at', 'payment_status']));
      allow delete: if false;
    }

    match /inventory/{itemId} {
      allow read: if isActiveUser();
      allow create: if isOwner();
      allow update: if isOwner()
        || (hasInventoryAccess()
          && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['quantity', 'status', 'last_updated']));
      allow delete: if false;
    }

    match /stock_movements/{moveId} {
      allow read: if isActiveUser();
      allow create: if hasInventoryAccess()
        && request.resource.data.created_by == request.auth.uid;
      allow update, delete: if false;
    }

    match /buffet_tracking/{rowId} {
      allow read: if isOwner();
      allow write: if isOwner();
    }

    match /expenses/{expenseId} {
      allow read, create, update: if isOwner();
      allow delete: if false;
    }

    match /audit_logs/{logId} {
      allow read: if isOwner();
      allow create: if isActiveUser()
        && request.resource.data.actor_id == request.auth.uid;
      allow update, delete: if false;
    }

    match /daily_summaries/{dayId} {
      allow read: if isOwner();
      allow create, update: if isActiveUser();
      allow delete: if false;
    }

    match /settings/{id} {
      allow read: if isActiveUser();
      allow write: if isOwner();
    }

    match /counters/{id} {
      allow read: if isActiveUser();
      allow write: if isActiveUser();
    }
  }
}
